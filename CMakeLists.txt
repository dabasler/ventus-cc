cmake_minimum_required(VERSION 3.16)
project(ventus-cc VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ventus_core STATIC core/src/Engine.cpp)
target_include_directories(ventus_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core/include)
if(MSVC)
  target_compile_options(ventus_core PRIVATE /W4)
else()
  target_compile_options(ventus_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ventus-cli cli/main.cpp)
target_link_libraries(ventus-cli PRIVATE ventus_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(ventus-tests tests/test_engine.cpp)
  target_link_libraries(ventus-tests PRIVATE ventus_core)
  add_test(NAME ventus-core COMMAND ventus-tests)
endif()
